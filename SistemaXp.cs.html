1    using System;
2    using RPG.Classes;
3    
4    namespace RPG.Systems
5    {
6        public class SistemaXP
7        {
8            public int XpAtual { get; private set; } = 0;
9            private SistemaNivel _sistemaNivel;
10   
11           public SistemaXP(SistemaNivel sistemaNivel)
12           {
13               _sistemaNivel = sistemaNivel;
14           }
15   
16           public void GanharXP(int quantidade, Classe jogador)
17           {
18               XpAtual += quantidade;
19               Console.WriteLine($"Você ganhou {quantidade} de XP! ({XpAtual}/{_sistemaNivel.XpParaProximoNivel})");
20   
21               // Caso o XP ganho passe do limite, ele sobe de nível (pode subir mais de um se ganhar muita XP)
22               while (XpAtual >= _sistemaNivel.XpParaProximoNivel)
23               {
24                   XpAtual -= _sistemaNivel.XpParaProximoNivel;
25                   _sistemaNivel.SubirDeNivel(jogador);
26               }
27           }
28       }
29   }
